// 1) Realtime Database Rules (Firebase Console > Build > Realtime Database > Rules):
//    Neeche wala JSON wahan paste karein aur Publish dabayein.
//    Matlab: customer bina login ke order likh sakta hai aur settings parh
//    sakta hai; settings BADALNA sirf login kiya hua shopkeeper kar sakta
//    hai; orders parhna/likhna sab ko (customer + dashboard + print agent).
//    Counter sirf order number ke liye hai.
//    notifyTokens: dashboard app (PWA) apna FCM token yahan likhta hai
//    (login ke baad), aur print agent yahan se token parh kar push
//    bhejta hai; ghalat token ko agent yahan se delete bhi karta hai.
//    Is liye read/write sirf login kiya hua (admin/agent) kar sakta hai.
//    agentStatus: print agent (v8.3 heartbeat) har 30 second mein apni
//    lastSeen likhta hai (sirf login kiya hua agent likh sakta hai);
//    parhna sab ko hai - customer page ka dot bina login ke dekhta hai,
//    dashboard bhi (login ke baad). Is liye read sab, write auth only.
// ============================================================================
{
  "rules": {
    "shops": {
      "$shopId": {
        "settings": {
          ".read": true,
          ".write": "auth != null"
        },
        "orders": {
          ".read": true,
          ".write": true
        },
        "counter": {
          ".read": true,
          ".write": true
        },
        "notifyTokens": {
          ".read": "auth != null",
          ".write": "auth != null"
        },
        "agentStatus": {
          ".read": true,
          ".write": "auth != null"
        }
      }
    }
  }
}
